```ts
/** What a stored blob belongs to. checksum/extension feed the external store's key (Phase 17/18); the inline store addresses by id. */
export interface BlobRef {
  object: string;          // "ContentVersion"
  field: string;           // "VersionData"
  id: string;              // 18-character record Id
  checksum: string | null; // lowercase MD5 hex, as stored in Checksum
  extension: string | null;// FileExtension
}
export interface OpenedBlob {
  stream: Readable;
  /** Bytes the store will deliver; the authority for Content-Length. */
  size: number;
}
export interface BlobStore {
  readonly kind: "inline" | "external";
  /** Writes the bytes for ref. Inline: joins the caller's transaction through `client`. */
  put(client: Queryable, ref: BlobRef, bytes: Buffer): Promise<void>;
  /** undefined when no bytes are behind ref (never an empty stream). */
  open(client: Queryable, ref: BlobRef): Promise<OpenedBlob | undefined>;
}
export class InlineBlobStore implements BlobStore { constructor(store: Store) }
export function blobKey(checksum: string, extension: string): string;
```
</interfaces>

Decisions written into this plan (planner, from research open questions):
- `blobKey` with an empty extension returns `files/<cs[0:4]>/<cs>` (no trailing dot). HEIMDALL-FORMAT.md only states
  `<cs>.<FileExtension>`; Phase 16/18 must confirm against Heimdall's S3Service before freezing the fixture layout. Put
  this note in the doc comment of `blobKey`.
- `.jpeg` and `.json` are NOT in the table (no public Salesforce source for a `JPEG`-for-.jpeg or `JSON` token; D-09 says
  leave unsourced tokens out) -> `UNKNOWN` / `application/octet-stream`.
- `FileExtension` is the lowercased text after the last `.` of the `PathOnClient` basename (orglet choice, MEDIUM: the
  Heimdall key and Salesforce exports show lowercase extensions); no dot, a leading-dot-only name (`.bashrc`) or a
  trailing dot -> `null`. Truncate to 40 characters (the field's length).
