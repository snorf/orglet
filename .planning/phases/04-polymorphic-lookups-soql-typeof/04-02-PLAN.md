    ```typescript
    /**
     * TYPEOF restrictions from the SOQL and SOSL Reference ("TYPEOF" considerations). Some invalid
     * forms already fail in the parser with a generic message and some parse; both end as
     * MALFORMED_QUERY whose message names the documented restriction (POLY-06).
     */
    import type { Query } from "@jetstreamapp/soql-parser-js";
    import { malformed } from "./errors.js";

    export const TYPEOF_RESTRICTIONS = {
      selectOnly: "TYPEOF is only allowed in the SELECT clause of a query",
      noObjects: "TYPEOF isn't allowed in queries that don't return objects, such as COUNT()",
      nested: "TYPEOF expressions can't be nested",
      semiJoin: "TYPEOF isn't allowed in the SELECT clause of a semi-join query",
      functions: "TYPEOF can't be used in queries with functions in the SELECT clause",
      grouping: "TYPEOF can't be used in queries with GROUP BY, GROUP BY ROLLUP, GROUP BY CUBE, and HAVING",
    } as const;

    export const alsoSelected = (relationship: string): string =>
      `TYPEOF ${relationship}: a relationship field used in TYPEOF can't also be referenced in the field list of the SELECT statement`;
    ```
    `export function diagnoseTypeofParseError(soql: string): string | undefined` — only called when parsing failed and `/\bTYPEOF\b/i` matches. Algorithm (no hand-rolled SOQL parser, text heuristic only):
      a. `text = soql.replace(/'(?:\\.|[^'\\])*'/g, "''")` (blank string literals).
      b. `m = /\bTYPEOF\b/i.exec(text)`; none -> `undefined`.
      c. `before = text.slice(0, m.index)`; strip balanced parenthesised groups repeatedly: `while (/\([^()]*\)/.test(before)) before = before.replace(/\([^()]*\)/g, " ");` so keywords inside earlier subqueries don't count.
      d. `clause` = the last match of `/\b(SELECT|WHERE|GROUP\s+BY|HAVING|ORDER\s+BY)\b/gi` in `before`, upper-cased with whitespace collapsed to one space.
      e. `GROUP BY` or `HAVING` -> `TYPEOF_RESTRICTIONS.grouping`; `WHERE` or `ORDER BY` -> `TYPEOF_RESTRICTIONS.selectOnly`.
      f. Otherwise (SELECT clause): `rest = text.slice(m.index + m[0].length)`; `span` = `rest` up to the first `/\bEND\b/i` (all of `rest` if none). If `/\bTYPEOF\b/i.test(span)` -> `nested`. Else if `/\b(THEN|ELSE)\b[\s\S]*\(/i.test(span)` -> `functions`. Else `undefined`.
    `export function assertTypeofAllowed(query: Query): void` — returns immediately when `!(query.fields ?? []).some((f) => f.type === "FieldTypeof")`. Otherwise, in this order:
      1. any field `f.type === "FieldFunctionExpression" && f.functionName.toUpperCase() === "COUNT" && f.parameters.length === 0` -> `throw malformed(TYPEOF_RESTRICTIONS.noObjects)`;
      2. `query.groupBy !== undefined || query.having !== undefined` -> `grouping`;
      3. any other `FieldFunctionExpression` -> `functions`;
      4. for each `FieldTypeof` `t`, let `rel = t.field.toLowerCase()`; if any `FieldRelationship` `r` has `r.relationships.join(".").toLowerCase() === rel` or starts with `${rel}.` -> `throw malformed(alsoSelected(t.field))`.
    Use only `malformed()` from errors.ts (no new error class).
    3. Wire into `packages/soql/src/compile.ts` (import `{ TYPEOF_RESTRICTIONS, assertTypeofAllowed, diagnoseTypeofParseError } from "./typeof.js"`):
      - catch block: `const detail = (err as Error).message; throw malformed(/\bTYPEOF\b/i.test(soql) ? (diagnoseTypeofParseError(soql) ?? detail) : detail);`
      - immediately after `if (!query.sObject) throw malformed("unexpected token: FROM");` add `assertTypeofAllowed(query);` (before the WITH check and before `countOnly`/`aggregate` at lines 182-183).
