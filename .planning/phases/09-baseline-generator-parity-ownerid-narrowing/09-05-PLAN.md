function isWidening(from: string, to: string): boolean {
  if (from === to) return true;
  const f = parseSized(from); const t = parseSized(to);
  if (f && t && f.base === t.base) { if (f.base === "numeric") return t.a - t.b >= f.a - f.b && t.b >= f.b; return t.a >= f.a; }
  if (f?.base === "varchar" && to === "text") return true;
  return false;
}
